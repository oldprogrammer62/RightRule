<expr>       ::= <or>

<or>         ::= <and> ( "or" <and> )*

<and>        ::= <not> ( "and" <not> )*

<not>        ::= "not" <not> 
               | <primary>

<primary>    ::= "(" <expr> ")"
               | <condition>

<condition>  ::= <left_operand> <operator> <right_operand>

<left_operand>    ::= <data_reference> | <value>
<right_operand>   ::= <data_reference> | <value>

<data_reference>  ::= <identifier_string> "." <identifier_string>

<identifier_string> ::= [a-zA-Z_][a-zA-Z0-9_]*

<value>           ::= <number> | <string> | <boolean> | <date> | <datetime>

<operator>   ::= "contains", "equals", "notequals"
// Custom operators can be included
// operator as evaluation command is case sensitive
