<expr>       ::= <or>

<or>         ::= <and> ( "or" <and> )*

<and>        ::= <not> ( "and" <not> )*

<not>        ::= "not" <not> 
               | <primary>

<primary>    ::= "(" <expr> ")"
               | <condition>

<condition>  ::= <left_operand> <operator> <right_operand>

<left_operand>    ::= <variable> | <value>
<right_operand>   ::= <variable> | <value>

<operator>   ::= "contains", "equals", "notequals", etc
// operator as evaluation command is case sensitive

<operator>   ::= "and", "or", "not"
// operator as logical command is lower case sensitive

<identifier>  ::= <variable> | <variable>.<variable>
// identifier is case sensitive - discovery in arguments

<variable>   ::= data pointer
// variable as data pointer is case sensitive - discovery in data
// Before . Data name
// After  . Data item name

<value>      ::= number, string, boolean
