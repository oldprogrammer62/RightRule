<expr>       ::= <or>

<or>         ::= <and> ( "OR" <and> )*

<and>        ::= <not> ( "AND" <not> )*

<not>        ::= "NOT" <not> 
               | <primary>

<primary>    ::= "(" <expr> ")"
               | <condition>

<condition>  ::= <variable> <operator> <value>

<operator>   ::= "contains", "equals", "notequals", etc

<value>      ::= number, string, boolean, placeholder

<placeholder> ::= "{" identifier "}"
