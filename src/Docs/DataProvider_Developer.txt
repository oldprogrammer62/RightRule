The Data Adaptor that transforms BespokeClass data into the required data structure.
PoviderDataStructure will be ingested by RightRule

internal class BespokeDataAdaptor : DataAdaptorBase<BespokeClass>
{
    public override AdaptorDataStructure Process(string DatasetName, BespokeClass input)
    {
        var result = new AdaptorDataStructure();
        result.DatasetName = DatasetName;
        result.Data.Add("Id", input.Id);
        result.Data.Add("Name", input.Name);
        result.Data.Add("Value", input.Value);
        return result;
    }
}
