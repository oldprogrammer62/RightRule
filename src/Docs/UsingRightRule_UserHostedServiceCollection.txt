
     ServiceCollection? services = new ServiceCollection();

     // Configure and set up Custom Logger
     var configuration = new ConfigurationBuilder()
         .SetBasePath(AppContext.BaseDirectory)
         .AddJsonFile("appsettings.json", optional: true)
         .Build();
     services.AddSingleton<IConfiguration>(configuration);

     services.AddLogging(builder =>
     {
         builder.AddRightRuleLogger(); // Custom Logger?
     });

     RightRuleOptions rightRuleOptions = new RightRuleOptions

            DataCheck = false,
            Diagnostics = true,
            LoadDefaultOperators = true,
            LoadCustomOperators = true,
            // CustomOperatorLibraryPath = ""
        };

     // Register RightRule and associated services
     services.AddRightRule(rightRuleOptions);

     return services;
