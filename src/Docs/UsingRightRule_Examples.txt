Manually instantiate RightRule
        IRightRule rr = new RightRule.RightRule();
With Options
        RightRuleOptions options = new RightRuleOptions { LoadDefaultOperators = true };
        IRightRule rr = new RightRule.RightRule(options);


Using RightRule with a self hosted Dependency Injection service collection
        var services = new ServiceCollection();
        services.AddLogging();
        services.AddRightRule();
        using var serviceProvider = services.BuildServiceProvider();
        using var scope = serviceProvider.CreateScope();
        var rr = scope.ServiceProvider.GetRequiredService<RightRule.IRightRule>();
With Options
        var services = new ServiceCollection();
        services.AddLogging();
        services.AddRightRule(options => options.LoadDefaultOperators = true);
        using var serviceProvider = services.BuildServiceProvider();
        using var scope = serviceProvider.CreateScope();
        var rr = scope.ServiceProvider.GetRequiredService<RightRule.IRightRule>();


Using RightRule with a system (asp.net) hosted Dependence Injection service collection
        builder.Services.AddRightRule();
With Options
        builder.Services.AddRightRule(options => options.LoadDefaultOperators = true);
