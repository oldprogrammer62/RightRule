using Microsoft.Extensions.DependencyInjection;
using RightRule;
using RightRule.LanguageProcessor.Evaluator;
using RightRule.RuleServices.RuleModels;
using RightRule.RuleServices.RuleModels.BaseModels;
using RightRule.ServiceRegister;

public static class OrderDiscountRegistration
{
    public static IServiceCollection AddOrderDiscountExample(
        this IServiceCollection services)
    {
        services.AddLogging();
        services.AddRightRule(options =>
        {
            options.DataCheck = false;
            options.Diagnostics = true;
            options.LoadCustomOperators = true;
            options.LoadDefaultOperators = true;
        });
        services.AddScoped<OrderDiscountExample>();
        return services;
    }
}

public sealed class OrderDiscountExample
{
    private readonly IRightRule _rr;

    public OrderDiscountExample(IRightRule rr) => _rr = rr;

    public RightRuleResult<IProcessedRule>? ApplyOrderDiscount(
        decimal orderValue,
        int previousOrders)
    {
        // The 2 parameters drive the rules process.
        // They describe the customers current order situation
        //  Order Value
        //  Previous Orders

        // This informs RightRule the event that is being processed.
        const string context = "OrderDiscount";

        rr.RegisterData("Order", new Dictionary<string, object>
        {
            ["Value"] = orderValue,
            ["PreviousOrders"] = previousOrders
        });

        rr.RegisterData("OrderThresholds", new Dictionary<string, object>
        {
            ["LoyaltyValue"] = 100m,
            ["ReturningValue"] = 50m,
            ["LoyaltyPreviousOrders"] = 5,
            ["ReturningPreviousOrders"] = 2
        });

        rr.RegisterData("Discount", new Dictionary<string, object>
        {
            ["Default"] = 2,
            ["GoodWill"] = 5,
            ["Returning"] = 10,
            ["Loyalty"] = 15
        });

        var newDefaultRule = new DefaultRule()
        {
            Name = "DefaultCustomer",
            Context = context,
            Decision = "NoDiscount",
            DecisionData = """{"discount":"$Discount.Default%"}"""
        };
        rr.RegisterDefaultRule(newDefaultRule);

        rr.RegisterRules(new List<DefinedRule>
        {
            new()
            {
                Number = 1,
                Name = "NewCustomer",
                Context = context,
                Statement = "Order.Value >= OrderThresholds.ReturningValue and Order.PreviousOrders < OrderThresholds.ReturningPreviousOrders",
                Decision = "Discount",
                DecisionData = """{"discount":"$Discount.GoodWill%"}""",
                FalseDecision = "Ineligible"
            },
            new()
            {
                Number = 2,
                Name = "LoyalCustomer",
                Context = context,
                Statement = "Order.Value >= OrderThresholds.LoyaltyValue and Order.PreviousOrders >= OrderThresholds.LoyaltyPreviousOrders",
                Decision = "Discount",
                DecisionData = """{"discount":"$Discount.Loyalty%"}""",
                FalseDecision = "Ineligible"
            },
            new()
            {
                Number = 3,
                Name = "ReturningCustomer",
                Context = context,
                Statement = "Order.Value >= OrderThresholds.ReturningValue and Order.PreviousOrders >= OrderThresholds.ReturningPreviousOrders",
                Decision = "Discount",
                DecisionData = """{"discount":"$Discount.Returning%"}""",
                FalseDecision = "Ineligible"
            }
        });
        return _rr.ApplyRule(context);
    }
}
