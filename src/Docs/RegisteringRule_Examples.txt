You can configure each rule with a specific "decision word" and "decision data" 
for both true and false outcomes.

Decision Word:
Tells the calling application which decision was made so it can route the next logical step.

Decision Data: 
Returns additional, context-specific information that the calling application can use during downstream processing.







Initialise a Rule Definition and Register
-----------------------------------------
IRightRule rr = new RightRule.RightRule();
RuleConfiguration rc = new RuleConfiguration()
{
    DataCheck = true,
    Diagnostics = false
};

RuleDefinition rule = new()
{
    Number = 1,
    Name = "Rule1",
    Context = "Context",
    Statement = "Test.boolvalue equals true",
    Decision = "EQUALS",
    DecisionData = null,
    OptionalData = null,
    RuleConfiguration = rc
};
RuleRegistrationResult rule1 = rr.RegisterRule(rule);


Build a Rule Definition and Register
------------------------------------
IRightRule rr = new RightRule.RightRule();
RuleDefinition rule = rr.BuildRuleDefinition()
   .Number(1)
   .Name("Rule1")
   .Context("Context")
   .Statement("Test.boolvalue equals true")
   .Decision("EQUALS")
   .DecisionData(null)
   .OptionalData(null)
   .WithCheckData(false)
   .WithDiagnostics(false)
   .Build();
RuleRegistrationResult rule1 = rr.RegisterRule(rule);
