Local intrinsic variables
-------------------------
- type.IsPrimitive
- propertyValue is string
- propertyValue is decimal
- propertyValue is DateTime
- propertyValue is DateOnly
- propertyValue is TimeOnly

// Declare variable
long localVal = 1000;

// Register Data
// Registration name "localData"
DataRegistrationResult registeredData = rr.RegisterData("localData", localVal);

// Rule Statement refers to Registered name "localData"
Statement = "localData = 1000"


Dictionary
----------
// Declare dictionary
Dictionary<string, object> data = new Dictionary<string, object>()
{
     { "entry", 1000 }
};

// Register Data
// Registration name "localData"
DataRegistrationResult registeredData = rr.RegisterData("localData", data);

// Rule statement refers to Registered name "localData" with registered value "entry"
Statement = "localData.entry = 1000"
