<expr>            ::= <or>

<or>              ::= <and> ( "or" <and> )*

<and>             ::= <not> ( "and" <not> )*

<not>             ::= "not" <not> | <primary>

<primary>         ::= "(" <expr> ")" | <condition>

<condition>       ::= <left_operand> <operator> <right_operand>

<operator>        ::= "contains", "equals", "notequals"

<left_operand>    ::= <data_reference> | <variable> | <value>
<right_operand>   ::= <data_reference> | <variable> | <value>

<data_reference>  ::= <string> "." <string>

<variable>        ::= <string>

<value>           ::= <number> | <string> | <boolean> | <date> | <datetime> | <array>

<number>          ::= <digits> ( "." <digits> )?
<digits>          ::= [0-9]+

<string>          ::= [a-zA-Z_][a-zA-Z0-9_]*

<boolean>         ::= true | false

<array>           ::= "[" (<number> ("," <number>)* | <string_literal> ("," <string_literal>)*) "]"
<string_literal>  ::= single-quoted or double-quoted string
